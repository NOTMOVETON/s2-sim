    // HTTP REST сервер управления симуляцией.
    // Команды → engine->enqueue(KernelCommand).
    // GET запросы читают состояние из engine напрямую (eventual consistency).
    class RestApiServer {

        RestApiServer(int port, SimEngine* engine, std::string scenes_dir = "");

        void run_server();

        // Парсинг HTTP: возвращает {method, path, body}
        struct HttpRequest { std::string method, path, body; };
        static HttpRequest parse_request(const std::string& raw);

        // HTTP ответы
        static void send_json(int fd, int status, const std::string& body);
        static void send_ok(int fd);
        static void send_error(int fd, int status, const std::string& msg);

        // Роутинг
        void dispatch(int fd, const HttpRequest& req);

        // Обработчики sim control
        void handle_sim_pause(int fd);
        void handle_sim_resume(int fd);
        void handle_sim_reset(int fd);

    Создать workspace/s2_visualizer/src/rest_api_server.cpp.

    СТРУКТУРА СЕРВЕРА (следовать паттерну из viz_server.cpp):
    - `start()`: открыть сокет (socket/bind/listen), запустить thread_(run_server)
    - `stop()`: running_=false, закрыть server_fd_, join thread
    - `run_server()`: accept loop → handle_client в отдельном detached thread
    - `handle_client(fd)`: прочитать запрос, parse_request, dispatch, закрыть fd

    `parse_request()`:
    - Читает первую строку: "METHOD /path HTTP/1.1"
    - Читает заголовки до пустой строки
    - Читает body (по Content-Length если есть)
    - Возвращает HttpRequest{method, path, body}

    `dispatch()` — роутинг по method+path:
    ```
    POST /sim/pause         → handle_sim_pause
    POST /sim/resume        → handle_sim_resume
    POST /sim/reset         → handle_sim_reset
    POST /sim/step          → handle_sim_step
    POST /sim/speed         → handle_sim_speed
    GET  /sim/status        → handle_sim_status
    GET  /world/entities    → handle_entities_list
    POST /world/entities    → handle_entity_spawn
    DELETE /world/entities/{id} → handle_entity_despawn
    PUT  /world/entities/{id}/pose    → handle_entity_set_pose
    PUT  /world/entities/{id}/enabled → handle_entity_set_enabled
    GET  /world/zones        → handle_zones_list
    POST /world/zones        → handle_zone_spawn
    DELETE /world/zones/{id} → handle_zone_despawn
    PUT  /world/zones/{id}/enabled  → handle_zone_set_enabled
    PUT  /world/zones/{id}/strength → handle_zone_set_strength
    POST /agents/{id}/input/{plugin} → handle_agent_plugin_input
    GET  /scenes            → handle_scenes_list
    POST /scenes            → handle_scene_load
    ```
    Все остальные → send_error(fd, 404, "not found")

    РЕАЛИЗАЦИЯ ОБРАБОТЧИКОВ (примеры):
    ```cpp
    void RestApiServer::handle_sim_pause(int fd) {
        engine_->enqueue(cmd::PauseSim{});
        send_ok(fd);
    }
    void RestApiServer::handle_sim_resume(int fd) {
        engine_->enqueue(cmd::ResumeSim{});
        send_ok(fd);
    }
    void RestApiServer::handle_sim_reset(int fd) {
        engine_->enqueue(cmd::ResetSim{});
        send_ok(fd);
    }
    void RestApiServer::handle_sim_step(int fd, const std::string& body) {
        int n = 1;
        try { auto j = nlohmann::json::parse(body); n = j.value("n", 1); } catch(...) {}
        engine_->enqueue(cmd::StepSim{n});
        send_ok(fd);
    }
    void RestApiServer::handle_sim_speed(int fd, const std::string& body) {
        double factor = 1.0;
        try { auto j = nlohmann::json::parse(body); factor = j.value("factor", 1.0); } catch(...) {}
        engine_->enqueue(cmd::SetSpeed{factor});
        send_ok(fd);
    }
    void RestApiServer::handle_sim_status(int fd) {
        nlohmann::json j;
        j["paused"] = engine_->is_paused();
        j["sim_time"] = engine_->sim_time();
        send_json(fd, 200, j.dump());
    }
    ```

    handle_entity_set_pose: парсить {"x":0,"y":0,"z":0,"yaw":0} → cmd::SetPose{id, pose}
    handle_entity_set_enabled: парсить {"enabled":true} → cmd::SetEnabled{id, enabled}
    handle_entity_spawn: парсить {type,name,pose} → cmd::SpawnEntity{id,type,name,pose}
    handle_entity_despawn: парсить id из пути → cmd::DespawnEntity{id}
    handle_zone_spawn: парсить {id,shape} → cmd::SpawnZone{...}
    handle_zone_despawn: id из пути → cmd::DespawnZone{id}
    handle_zone_set_enabled: {"enabled":bool} → cmd::ToggleZone{id,enabled}
    handle_zone_set_strength: {"strength":0.5} → cmd::SetZoneStrength{id,strength}
    handle_agent_plugin_input: парсить agent id + plugin из пути, body → engine_->handle_plugin_input(id, plugin, body)
    handle_scenes_list: сканировать scenes_dir_ → JSON {"scenes":[...]}
    handle_scene_load: {"filename":"test.yaml"} → cmd::LoadScene{scenes_dir_ + "/" + filename}

    send_ok(fd): "HTTP/1.1 200 OK\r\nContent-Type: application/json\r\n\r\n{\"ok\":true}"
    send_json(fd, status, body): HTTP ответ с body
    send_error(fd, status, msg): {"error":"msg"}

    CORS заголовки: добавить "Access-Control-Allow-Origin: *" ко всем ответам.
    OPTIONS запросы: отвечать 200 с CORS preflight заголовками.

    Avoid: блокировать engine на время обработки GET запросов — читать state без мьютекса
    (acceptable: eventual consistency для read-only запросов)
    Avoid: держать клиентский сокет открытым дольше нужного — закрывать после каждого запроса
