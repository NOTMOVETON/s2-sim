TEST(TickLifecycle, ClearContributionsOnlyInPhase8)
{
  // Если бы clear_contributions вызывался в Phase 3 (как раньше),
  // то contributions, выставленные в pre_resolve Phase 3a,
  // были бы очищены ДО следующего тика — нормально.
  // Но если плагин SENSOR в Phase 4 зависит от effective() после resolve —
  // он должен видеть непоменявшееся состояние.
  //
  // Этот тест проверяет что clear_contributions() НЕ вызывается между Phase 3 и Phase 4:
  // плагин RESOURCE кладёт contribution в pre_resolve,
  // плагин SENSOR читает effective() в update() и получает ненулевое значение.

  SimEngine::Config cfg;
  cfg.update_rate = 100.0;
  cfg.viz_rate    = 0.0;
  cfg.transport_rate = 0.0;
  SimEngine engine(cfg);

  bool sensor_saw_scale = false;

  struct ResourcePlugin : public plugins::IAgentPlugin {
    std::string type() const override { return "resource_test"; }
    plugins::PluginRole role() const override { return plugins::PluginRole::RESOURCE; }
    void pre_resolve(double, Agent& agent) override {
      agent.state.add_scale(0.5, "test_zone");
    }
    void update(double, Agent&, const plugins::PluginContext&) override {}
    void from_config(const YAML::Node&) override {}
    std::string to_json() const override { return "{}"; }
  };

  struct SensorPlugin : public plugins::IAgentPlugin {
    explicit SensorPlugin(bool& out) : out_(out) {}
    std::string type() const override { return "sensor_test"; }
    plugins::PluginRole role() const override { return plugins::PluginRole::SENSOR; }
    void update(double, Agent& agent, const plugins::PluginContext&) override {
      // Если clear_contributions уже был вызван — effective().speed_scale == 1.0
      // Если нет — speed_scale == 0.5 (как выставил ResourcePlugin)
      out_ = (std::abs(agent.state.effective().speed_scale - 0.5) < 1e-6);
    }
    void from_config(const YAML::Node&) override {}
    std::string to_json() const override { return "{}"; }
    bool& out_;
  };

  Agent agent;
  agent.id = 1;
  agent.plugins.push_back(std::make_unique<ResourcePlugin>());
  agent.plugins.push_back(std::make_unique<SensorPlugin>(sensor_saw_scale));

  SimWorld world;
  world.agents().push_back(std::move(agent));
  engine.load_world(std::move(world));

  engine.step(1);

  EXPECT_TRUE(sensor_saw_scale)
      << "Sensor должен видеть speed_scale=0.5, выставленный ResourcePlugin в pre_resolve. "
         "Если false — clear_contributions был вызван между Phase 3 и Phase 4.";
}
