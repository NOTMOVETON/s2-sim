void ZoneSystem::update_lifecycle(double dt) {
    for (auto& zone : zones_) {
        // Рост
        if (zone.lifecycle.growth_rate > 0.0 && zone.strength < zone.lifecycle.max_strength) {
            zone.strength = std::min(zone.strength + zone.lifecycle.growth_rate * dt,
                                     zone.lifecycle.max_strength);
        }
        // Затухание (dt накапливается через sim_time - spawn_time, упрощённо через strength)
        // Для простоты: если decay_rate > 0 и strength <= initial — начинаем убывать
        if (zone.lifecycle.decay_rate > 0.0) {
            zone.strength = std::max(0.0, zone.strength - zone.lifecycle.decay_rate * dt);
        }
    }
}

bool ZoneSystem::agent_in_zone(const Agent& agent, const Zone& zone) {
    switch (zone.detection_mode_enum) {
    case DetectionMode::CENTER: {
        Vec3 pos = agent.world_pose.position();
        return zone_contains(zone.shape, pos);
    }
    case DetectionMode::BOUNDING: {
        // Упрощённый AABB/sphere overlap
        Vec3 pos = agent.world_pose.position();
        double agent_r = 0.5; // дефолт если нет bounding shape
        if (zone.shape.type == ZoneShapeType::SPHERE) {
            return (pos - zone.shape.center).norm() < (zone.shape.radius + agent_r);
        }
        // Для AABB box: clamp point, check distance
        return zone_contains(zone.shape, pos); // fallback
    }
    case DetectionMode::PER_LINK: {
        if (!agent.kinematic_tree) {
            // Fallback на CENTER если нет URDF
            return zone_contains(zone.shape, agent.world_pose.position());
        }
        // Итерация линков — прочитать KinematicTree API
        // for each link: если link world_pos в зоне → contact_link = link.name; return true
        // КОНКРЕТНУЮ РЕАЛИЗАЦИЮ построить на основе прочитанного kinematic_tree.hpp
        return zone_contains(zone.shape, agent.world_pose.position()); // placeholder
    }
    }
    return false;
}

void ZoneSystem::update_owned_zones_positions(const std::vector<Agent>& agents) {
    for (auto& zone : zones_) {
        if (zone.attached_to_entity_id.empty()) continue;
        // Найти агента по id
        const Agent* owner = nullptr;
        for (const auto& agent : agents) {
            if (std::to_string(agent.id) == zone.attached_to_entity_id) {
                owner = &agent;
                break;
            }
        }
        if (!owner) continue;

        Vec3 base_pos = owner->world_pose.position();

        // Если задан attached_to_link — найти линк в kinematic_frames
        if (zone.attached_to_link.has_value() && owner->kinematic_tree) {
            // Прочитать KinematicTree API для итерации линков
            // Найти линк по имени, взять world_pose.position()
            // Если не найден — использовать base_pos
        }

        zone.shape.center = base_pos + zone.attachment_offset;
    }
}
