// struct WorldContext — контекст, передаваемый в behavior.update()

  const WorldQuery&   world;    // read-only доступ к миру

// struct SignalEvent — событие сигнала для on_signal()

// class IActorBehavior — базовый интерфейс поведения актора
class IActorBehavior {
public:
  virtual ~IActorBehavior() = default;
  virtual std::string type() const = 0;
  virtual void on_init(const YAML::Node& config) = 0;
  virtual void on_spawn(ActorId actor_id) {}
  virtual void on_reset() {}
  virtual void update(double dt, Actor& actor, const WorldContext& ctx) = 0;
  virtual void on_signal(const SignalEvent& event) {}
  virtual void on_interact(EntityId source, const std::string& action,
                           const nlohmann::json& params) {}
  virtual std::string current_state() const = 0;
  virtual std::string to_json() const = 0;

  // Стабы Phase 7 (MaterialTransfer / DeformEntity) — возвращают false/null
  virtual bool can_release_material() const { return false; }
  virtual bool can_accept_material() const { return false; }

};

**actor_fsm.hpp** — утилитарный класс ActorFSM (per D-02):

```cpp
// ActorFSM — утилитарный класс конечного автомата для использования в IActorBehavior.
// DoorBehavior использует его, DirtPile (Phase 7) — нет.
// Хранит состояния как строки, transition-таблицу, текущее состояние.
class ActorFSM {
public:
  using OnEnter  = std::function<void()>;
  using OnUpdate = std::function<void(double dt)>;
  using OnExit   = std::function<void()>;
  using Guard    = std::function<bool()>;

  struct State {
    std::string name;
    OnEnter     on_enter;   // nullptr = no-op
    OnUpdate    on_update;  // nullptr = no-op
    OnExit      on_exit;    // nullptr = no-op
  };

  struct Transition {
    std::string from;
    std::string to;
    std::string trigger;
    Guard       guard;  // nullptr = always allowed
  };

  void add_state(std::string name, OnEnter on_enter = nullptr,
                 OnUpdate on_update = nullptr, OnExit on_exit = nullptr);
  void add_transition(std::string from, std::string to,
                      std::string trigger, Guard guard = nullptr);
  bool fire(const std::string& trigger);  // возвращает true если переход выполнен
  void update(double dt);                 // вызывает on_update текущего состояния
  const std::string& current_state() const { return current_; }

private:
  std::vector<State>      states_;
  std::vector<Transition> transitions_;
  std::string             current_;
};
```

Реализацию методов ActorFSM (add_state, add_transition, fire, update) разместить прямо в заголовке как inline-методы — FSM небольшой, нет смысла выносить в .cpp.

`fire()`:
- Найти переход где from == current_ и trigger == trigger
- Если guard задан и guard() == false — не переходить, вернуть false
- Найти State с именем to, вызвать on_exit текущего, установить current_ = to, вызвать on_enter нового
- Вернуть true

`update(dt)`:
- Найти State с именем current_, вызвать on_update(dt) если не nullptr

`add_state(name, ...)`:
- Добавить State в states_; если current_ пуст — установить current_ = name
