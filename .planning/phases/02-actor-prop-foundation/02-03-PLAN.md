**door_behavior.hpp:**
```cpp
#pragma once
#include <s2/actor_behavior.hpp>
#include <s2/actor_fsm.hpp>

namespace s2 {

class DoorBehavior : public IActorBehavior {
public:
  std::string type() const override { return "door"; }
  void on_init(const YAML::Node& config) override;
  void on_spawn(ActorId actor_id) override;
  void on_reset() override;
  void update(double dt, Actor& actor, const WorldContext& ctx) override;
  void on_signal(const SignalEvent& event) override;
  void on_interact(EntityId source, const std::string& action,
                   const nlohmann::json& params) override;
  std::string current_state() const override;
  std::string to_json() const override;

private:
  void setup_fsm();
  void on_enter_open(Actor& actor);
  void on_enter_closed(Actor& actor);

  ActorFSM fsm_;
  ActorId  actor_id_{0};

  // Конфигурация из YAML
  double open_duration_{1.0};   // секунды на открытие
  double close_duration_{1.0};  // секунды на закрытие
  double auto_close_secs_{0.0}; // 0 = не закрывается автоматически

  // Таймер открытия/закрытия
  double state_timer_{0.0};

  // Для публикации ActorStateChanged
  EventBus* bus_{nullptr};
};

} // namespace s2
```

**door_behavior.cpp:**

`on_init()`:
- Читает open_duration, close_duration, auto_close_secs из node с дефолтами
- Вызывает setup_fsm()

`setup_fsm()`:
- Добавляет состояния: "CLOSED", "OPENING", "OPEN", "CLOSING"
- Переходы:
  - CLOSED→OPENING trigger:"open"
  - OPENING→OPEN trigger:"_auto" (срабатывает таймером в update)
  - OPEN→CLOSING trigger:"close"
  - CLOSING→CLOSED trigger:"_auto"
- on_enter_open: actor.collision_enabled = false
- on_enter_closed: actor.collision_enabled = true
- Важно: on_enter коллбеки не могут принять `Actor&` напрямую через FSM без ссылки → хранить `Actor* current_actor_` и устанавливать в update() до fsm_.update()

Паттерн для collision в on_enter:
```cpp

```

`on_spawn(actor_id)`:
- Сохраняет actor_id_

`on_reset()`:
- fsm_.fire("close") если не CLOSED → принудительно CLOSED
- state_timer_ = 0

`update(dt, actor, ctx)`:
```cpp
current_actor_ = &actor;  // для on_enter коллбеков
bus_ = &ctx.bus;

state_timer_ += dt;
const auto& state = fsm_.current_state();

// Таймерные автопереходы
if (state == "OPENING" && state_timer_ >= open_duration_) {
  state_timer_ = 0;
  std::string old_state = state;
  fsm_.fire("_auto");
  ctx.bus.publish(event::ActorStateChanged{
      .actor = actor.id, .old_state = old_state,
      .new_state = fsm_.current_state()});
}
if (state == "CLOSING" && state_timer_ >= close_duration_) {
  state_timer_ = 0;
  std::string old_state = state;
  fsm_.fire("_auto");
  ctx.bus.publish(event::ActorStateChanged{
      .actor = actor.id, .old_state = old_state,
      .new_state = fsm_.current_state()});
}

// auto_close после OPEN
if (state == "OPEN" && auto_close_secs_ > 0 && state_timer_ >= auto_close_secs_) {
  state_timer_ = 0;
  std::string old_state = state;
  fsm_.fire("close");
  ctx.bus.publish(event::ActorStateChanged{
      .actor = actor.id, .old_state = old_state,
      .new_state = fsm_.current_state()});
}

fsm_.update(dt);
actor.current_state = fsm_.current_state();  // синхронизировать поле актора
current_actor_ = nullptr;
```

  void from_config(const YAML::Node& node) override;
