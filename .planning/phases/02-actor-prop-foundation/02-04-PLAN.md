**signal_listener_base.hpp** — размещается в `workspace/s2_plugins/include/s2/`:

```cpp
#pragma once
#include <s2/plugin_base.hpp>
#include <s2/event_bus.hpp>
#include <string>
#include <vector>

namespace s2 {

// SignalListenerBase — общая база для wire-контроллеров.
// Подписывается на SignalActivated/SignalDeactivated события при on_spawn().
// В update() позволяет унаследованным классам итерировать pending_signals_.
class SignalListenerBase : public plugins::IAgentPlugin {
public:
  // Вызвать в начале update() унаследованного класса
  // Очищает список pending_signals_ (события обработаны за этот тик)
  void flush_signals() { pending_signals_.clear(); }

  // Подписаться на EventBus при старте плагина
  void on_spawn(Agent& agent) override;

  // Список сигнальных событий, собранных с прошлого flush
  struct PendingSignal {
    std::string signal_id;
    EntityId    source_entity{0};
    bool        active{true};
  };
  const std::vector<PendingSignal>& pending_signals() const { return pending_signals_; }

protected:
  // Фильтрация по signal_id
  std::vector<PendingSignal> filter_by_id(const std::string& signal_id) const;
  // Фильтрация по source_entity
  std::vector<PendingSignal> filter_by_source(EntityId source_entity) const;

  EventBus* bus_{nullptr};  // устанавливается в on_spawn через ctx

private:
  std::vector<PendingSignal> pending_signals_;
};

} // namespace s2
```

Реализация on_spawn (inline в .hpp или в отдельном signal_listener_base.cpp):
```cpp
void SignalListenerBase::on_spawn(Agent& agent) {
  // Нет прямого доступа к EventBus из on_spawn(Agent&).
  // EventBus доступен через PluginContext в update().
  // Подписку делаем при первом update() — установить флаг подписки.
}
```

ПРОБЛЕМА: on_spawn() не имеет доступа к EventBus (он в PluginContext только при update()).
РЕШЕНИЕ: подписываться при первом вызове update(), используя флаг `subscribed_`:
```cpp
void SignalListenerBase::subscribe_once(EventBus& bus) {
  if (subscribed_) return;
  bus.subscribe<event::SignalActivated>([this](const event::SignalActivated& e) {
    pending_signals_.push_back({e.signal_id, e.source_entity, true});
  });
  bus.subscribe<event::SignalDeactivated>([this](const event::SignalDeactivated& e) {
    pending_signals_.push_back({e.signal_id, e.source_entity, false});
  });
  bus_ = &bus;
  subscribed_ = true;
}
```
Добавить protected `subscribe_once(EventBus& bus)` метод, вызываемый из update() каждого контроллера.

**door_wire_controller.hpp:**
```cpp
#pragma once
#include <s2/signal_listener_base.hpp>

namespace s2 {

// DoorWireController — плагин актора/агента.
// Слушает wire-сигналы и доставляет их в behavior.on_signal() актора.
// Конфигурация: reactions: [{signal_id, source_entity?, on_active, on_inactive}]
// on_active/on_inactive: "close_and_lock" / "force_open" / "unlock"
class DoorWireController : public SignalListenerBase {
public:
  std::string type() const override { return "door_wire_controller"; }
  PluginRole  role() const override { return PluginRole::INTERACTION; }

  void update(double dt, Agent& agent, const PluginContext& ctx) override;
  void from_config(const YAML::Node& node) override;
  std::string to_json() const override;

private:
  struct Reaction {
    std::string signal_id;
    EntityId    source_entity{0};  // 0 = любой источник
    std::string on_active;         // "close_and_lock", "force_open", "unlock", "open", "close"
    std::string on_inactive;       // "" = нет действия
  };
  std::vector<Reaction> reactions_;
};

} // namespace s2
```

`update(dt, agent, ctx)`:
```cpp
subscribe_once(ctx.bus);  // подписаться при первом вызове
flush_signals();          // НЕВЕРНО — flush нужен в конце, чтобы сигналы обработались

// ПРАВИЛЬНЫЙ порядок:
subscribe_once(ctx.bus);
// pending_signals_ заполнены из EventBus callbacks

for (const auto& reaction : reactions_) {
  for (const auto& sig : pending_signals()) {
    if (sig.signal_id != reaction.signal_id) continue;
    if (reaction.source_entity != 0 && sig.source_entity != reaction.source_entity) continue;

    const std::string& action = sig.active ? reaction.on_active : reaction.on_inactive;
    if (action.empty()) continue;

    // Доставить сигнал актору через Interact или напрямую через actor.behavior
    // В Phase 2 используем cmd::Interact чтобы ядро маршрутизировало
    // target_id = actor.id — но мы находимся на агенте или на акторе
    // РЕШЕНИЕ: actor плагины запускаются с actor.id как agent.id (из phase2_actors proxy)
    ctx.commands.push_back(cmd::Interact{
        .source_id    = agent.id,
        .target_id    = agent.id,  // актор сам себе — behavior.on_interact() или on_signal()
        .action       = action,
        .params       = nlohmann::json::object(),
        .max_distance = 0.0
    });
  }
}
flush_signals();  // очищаем после обработки
```

```cpp
#pragma once
#include <s2/signal_listener_base.hpp>
#include <nlohmann/json.hpp>

namespace s2 {

// EventReactor — декларативный плагин.
// Слушает wire-сигнал (или EventBus событие) → публикует другое событие.
// Конфигурация:
//   listen: {signal_id: "factory_power"}
//   on_active: {fire_event: "actor_state_changed", params: {actor: 5, state: "active"}}
//   on_inactive: {fire_event: "actor_state_changed", params: {actor: 5, state: "idle"}}
class EventReactor : public SignalListenerBase {
public:
  std::string type() const override { return "event_reactor"; }
  PluginRole  role() const override { return PluginRole::INTERACTION; }

  void update(double dt, Agent& agent, const PluginContext& ctx) override;
  void from_config(const YAML::Node& node) override;
  std::string to_json() const override;

private:
  std::string    listen_signal_id_;
  EntityId       listen_source_{0};   // 0 = любой источник

  // Действие при активации
  std::string    on_active_event_;    // тип события: "signal_activated", "actor_state_changed"
  nlohmann::json on_active_params_;

  // Действие при деактивации
  std::string    on_inactive_event_;
  nlohmann::json on_inactive_params_;
};

} // namespace s2
```
